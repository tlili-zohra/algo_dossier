algorithm Palandrome
var ch:string;
var i,j: integer;
begin
	readln("give me your word",ch) #madame =true
	i:=1;
	j:=length(ch)
	while((i<=length(ch)) and (j=>1))do
		if (ch[i]==ch[j]) then
			i++;
			j--;
		write("False");
		break;
	end_while
	write ("True");
end
