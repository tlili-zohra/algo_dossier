algorithm Seasons
	var num:Integer;
	begin
		read("Month number =",num);
		
		switch(num) do
			case 12,1,2 : write("winter");
			case 3,4,5: write("spring");
			case 6,7,8: write ("summer");
			case 9,10,11: write ("Fall");
			default: write("the nomber is not in the liste of season");
		endswitch
	end
