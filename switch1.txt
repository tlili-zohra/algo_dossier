algorithm check_numb
	var num:Integer;
	begin
		read("num =",num);
		switch(num) do
			case (num>0): write("Positive");
			case (num<0): write("negative");
			case (num==0): write ("Zero");
			default: write("this is not a number");
		endswitch
	end

switch(num) do
